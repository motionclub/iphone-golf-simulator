using System;
using System.Collections.Generic;
using System.IO;
using UnityEngine;

namespace GolfArcade.Tennis
{
    /// PROJECT-COPY ONLY (work/final-plan-1111/phase2/HeroStudio.cs.txt, put in place by install_clone_preview.sh): a preview of
    /// the studio's male body, tops and shoes in the real game, before anything is installed in the shared project.
    /// HERO_STUDIO=<dir> holds "<renderer name>_<vertex count>.pos" (float32 xyz per vertex, rest pose, hero frame; written by
    /// make_game_data.py). For the MALE hero, a mesh with a file takes those positions and gets no 11-11 build; a mesh without
    /// one is left exactly as it is. Normals and tangents are turned with the surface (welded face normals before and after),
    /// so authored seams and detail normals survive. She is untouched.
    public static class HeroStudio
    {
        public static string Dir => Environment.GetEnvironmentVariable("HERO_STUDIO");
        public static bool On => !string.IsNullOrEmpty(Dir);
        static readonly Dictionary<Mesh, Mesh> cache = new Dictionary<Mesh, Mesh>();

        /// `her`: the call comes after her own build (HeroBuild.Prepare), for her refitted tops; without it she is left alone.
        public static Mesh Override(MatchHeroLook look, SkinnedMeshRenderer smr, Mesh source, bool her = false)
        {
            if (!On || !look || (look.female && !her) || !smr || !source || !source.isReadable) return source;
            if (source.name.Contains("(studio)")) return source;
            if (cache.TryGetValue(source, out var done) && done) return done;
            int n = source.vertexCount; string file = Path.Combine(Dir, smr.name + "_" + n + ".pos");
            if (!File.Exists(file)) { Debug.Log($"[HeroStudio] {smr.name} {source.name} v={n}: no file, left as it is"); cache[source] = source; return source; }
            var bytes = File.ReadAllBytes(file);
            if (bytes.Length != n * 12) { Debug.LogWarning($"[HeroStudio] {file}: {bytes.Length} bytes for {n} vertices"); cache[source] = source; return source; }
            Matrix4x4 toHero = look.transform.worldToLocalMatrix * smr.transform.localToWorldMatrix, fromHero = toHero.inverse;
            var v = source.vertices; var after = new Vector3[n]; float largest = 0; int moved = 0;
            for (int i = 0; i < n; i++)
            {
                var p = new Vector3(BitConverter.ToSingle(bytes, i * 12), BitConverter.ToSingle(bytes, i * 12 + 4), BitConverter.ToSingle(bytes, i * 12 + 8));
                after[i] = fromHero.MultiplyPoint3x4(p); float d = (toHero.MultiplyPoint3x4(v[i]) - p).magnitude; if (d > 1e-5f) moved++; if (d > largest) largest = d;
            }
            // welded face normals of the surface before and after: each stored normal and tangent turns by the same rotation
            var weld = new Dictionary<Vector3Int, int>(); var group = new int[n]; int groups = 0;
            for (int i = 0; i < n; i++) { var key = new Vector3Int(Mathf.RoundToInt(v[i].x * 20000), Mathf.RoundToInt(v[i].y * 20000), Mathf.RoundToInt(v[i].z * 20000)); if (!weld.TryGetValue(key, out int g)) weld[key] = g = groups++; group[i] = g; }
            var was = new Vector3[groups]; var now = new Vector3[groups];
            for (int sm = 0; sm < source.subMeshCount; sm++)
            {
                var tris = source.GetTriangles(sm);
                for (int t = 0; t + 2 < tris.Length; t += 3)
                {
                    int a = tris[t], b = tris[t + 1], c = tris[t + 2];
                    var f0 = Vector3.Cross(v[b] - v[a], v[c] - v[a]); var f1 = Vector3.Cross(after[b] - after[a], after[c] - after[a]);
                    was[group[a]] += f0; was[group[b]] += f0; was[group[c]] += f0; now[group[a]] += f1; now[group[b]] += f1; now[group[c]] += f1;
                }
            }
            var normals = source.normals; var tangents = source.tangents;
            var newNormals = (Vector3[])normals.Clone(); var newTangents = (Vector4[])tangents.Clone();
            for (int i = 0; i < n; i++)
            {
                Vector3 a = was[group[i]], b = now[group[i]]; if (a.sqrMagnitude < 1e-20f || b.sqrMagnitude < 1e-20f) continue;
                var q = Quaternion.FromToRotation(a, b);
                if (i < normals.Length) newNormals[i] = (q * normals[i]).normalized;
                if (i < tangents.Length) { var t3 = (q * new Vector3(tangents[i].x, tangents[i].y, tangents[i].z)).normalized; newTangents[i] = new Vector4(t3.x, t3.y, t3.z, tangents[i].w); }
            }
            var ready = UnityEngine.Object.Instantiate(source); ready.name = source.name + " (studio)"; ready.hideFlags = HideFlags.DontSave;
            ready.vertices = after;
            if (normals.Length == n) ready.normals = newNormals;
            if (tangents.Length == n) ready.tangents = newTangents;
            ready.RecalculateBounds(); cache[source] = ready;
            Debug.Log($"[HeroStudio] {smr.name} {source.name} v={n}: moved={moved} largest={largest * 1000:F1}mm");
            return ready;
        }
    }
}
