using System.Collections;
using System.IO;
using GolfArcade.Game;
using GolfArcade.Tennis;
using NUnit.Framework;
using UnityEngine;
using UnityEngine.SceneManagement;
using UnityEngine.TestTools;

namespace GolfArcade.PlayTests
{
    /// Clone-only proof for the 2026-11-11 flat ball (copy to Assets/Tests/PlayMode/ZTennisFlightFilm.cs; TENNIS_FILM_OUT=<dir>).
    /// A real exhibition match on the gameplay camera, the bot swinging: a frame is saved at the top of each rally ball's
    /// flight (what the player sees when the ball is highest), and every ball's path is written out for the arc chart.
    /// Run once as it is and once with TENNIS_CLASSIC=1 for the before.
    public class ZTennisFlightFilm
    {
        [UnityTest, Timeout(1800000)]
        public IEnumerator ApexFramesAndArcs()
        {
            string dir = System.Environment.GetEnvironmentVariable("TENNIS_FILM_OUT") ?? "Library/Captures/flight";
            Directory.CreateDirectory(dir);
            LogAssert.ignoreFailingMessages = true;
            yield return SceneManager.LoadSceneAsync("Tennis"); yield return null;
            var game = Object.FindFirstObjectByType<TennisGame>();
            int oldRate = Time.captureFramerate; Time.captureFramerate = 60;
            try
            {
                for (int f = 0; f < (TennisPresentation.Length + 2) * 60; f++) yield return null;
                game.ConfigureMatch(TennisGame.Mode.Exhibition, null, "Rival", "CASUAL MATCH");
                game.AutoPlay = true; game.AutoPlayLean = true; TennisGame.AutoPlayTimingJitter = .06f;
                var arcs = new System.Text.StringBuilder("shot,byPlayer,t,x,y,z\n");
                int shot = 0, seen = -1, saved = 0, savedRival = 0, savedPlayer = 0; bool byPlayer = false, apexDone = true; float t0 = 0, lastVy = 0, clock = 0;
                for (int frame = 0; frame < 60 * 150 && (savedRival < 6 || savedPlayer < 6); frame++)
                {
                    yield return null; clock += 1f / 60;
                    if (game.Flow != TennisGame.Phase.Rally) { seen = -1; continue; }
                    if (game.RallyShots != seen)
                    {
                        seen = game.RallyShots; shot++; byPlayer = game.BallVelocity.z > 0; t0 = clock; apexDone = seen < 2;   // the serve is not a rally ball
                        lastVy = game.BallVelocity.y;
                    }
                    var p = game.BallPosition;
                    arcs.AppendLine(System.FormattableString.Invariant($"{shot},{(byPlayer ? 1 : 0)},{clock - t0:0.000},{p.x:0.00},{p.y:0.00},{p.z:0.00}"));
                    if (!apexDone && lastVy > 0 && game.BallVelocity.y <= 0)
                    {
                        apexDone = true;
                        if (byPlayer ? savedPlayer < 6 : savedRival < 6)
                        {
                            string name = System.FormattableString.Invariant($"{(byPlayer ? "player" : "rival")}_{(byPlayer ? savedPlayer : savedRival):00}_apex_{p.y:0.00}m.jpg");
                            GameCapture.Save(Path.Combine(dir, name), 1280, 720);
                            saved++; if (byPlayer) savedPlayer++; else savedRival++;
                        }
                    }
                    lastVy = game.BallVelocity.y;
                }
                File.WriteAllText(Path.Combine(dir, "arcs.csv"), arcs.ToString());
                Debug.Log($"[FlightFilm] {saved} apex frames, {shot} shots, classic={TennisFlight.Classic}");
                Assert.That(saved, Is.GreaterThan(4));
            }
            finally { game.AutoPlay = false; Time.captureFramerate = oldRate; }
        }
    }
}
