void Apply(bool initial=false){

void UpdateTargetWeights(float floor,float crouch) {

 for(int i=0;i<currentWeights.Length;i++) {
  Array.Clear(mass,0,mass.Length);var w=targetWeights[i];
  float knee=crouch*Mathf.SmoothStep(0,1,(i/Columns)/(float)(Rows-1))*(1-Mathf.SmoothStep(0,1,Mathf.InverseLerp(-.02f,.06f,rest[i].y)))*.9f;
  float source=(1-floor)*(1-knee);mass[w.boneIndex0]+=w.weight0*source;mass[w.boneIndex1]+=w.weight1*source;mass[w.boneIndex2]+=w.weight2*source;mass[w.boneIndex3]+=w.weight3*source;mass[hipIndex]+=floor*(1-knee);
  float left=Mathf.Clamp01(.5f-rest[i].x/.14f);mass[leftLegIndex]+=knee*left;mass[rightLegIndex]+=knee*(1-left);
  float sum=0;for(int k=0;k<4;k++){int best=0;for(int j=1;j<mass.Length;j++)if(mass[j]>mass[best])best=j;selected[k]=best;selectedMass[k]=mass[best];sum+=mass[best];mass[best]=0;}
  currentWeights[i]=new BoneWeight{boneIndex0=selected[0],boneIndex1=selected[1],boneIndex2=selected[2],boneIndex3=selected[3],weight0=selectedMass[0]/sum,weight1=selectedMass[1]/sum,weight2=selectedMass[2]/sum,weight3=selectedMass[3]/sum};
 }

}
void Project(ref Vector3 point,Vector3 a,Vector3 b,float radiusA,float radiusB,float allowance) {
 var axis=b-a;float t=axis.sqrMagnitude>1e-8f?Mathf.Clamp01(Vector3.Dot(point-a,axis)/axis.sqrMagnitude):0;

 if(delta.sqrMagnitude<radius*radius)point=center+(delta.sqrMagnitude>1e-10f?delta.normalized:transform.forward)*radius;
}
