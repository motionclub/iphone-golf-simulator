using System;
using System.Collections.Generic;
using UnityEngine;

namespace GolfArcade.Tennis
{
    /// GARMENT SHELL 2026-11-14 (Adnan: "his polo tears at the shoulder"). The kit tops are built as a thick double wall: an outer
    /// surface and, 2-15 mm inside it, an inner one facing the body (72 % of his polo's points have the other wall right behind them).
    /// Whenever an arm goes overhead (the serve, a high follow-through) the shoulder and armpit are squeezed and the inner wall pushes
    /// out through the outer one: jagged strips, the inside showing, skin in the gaps. Measured on 11-14: the same with the 11-11 tops
    /// (HERO_STUDIO=0) and at full detail (CHARACTER_LOD_FORCE=0), so neither the refit nor the LOD causes it.
    /// The inner wall is only ever seen through an opening (the hem, the cuffs, the collar), so it is kept within KeepNearRim of an
    /// edge of the wall and dropped everywhere else. Vertices and every attribute stay; only triangles go. Cached per source mesh.
    /// HERO_SHELL=0 (Editor) keeps the full double wall.
    public static class HeroGarmentShell
    {
        public static bool Enabled
        {
            get
            {
#if UNITY_EDITOR
                return Environment.GetEnvironmentVariable("HERO_SHELL") != "0";
#else
                return true;
#endif
            }
        }

        const float Weld = 1e-5f, Cell = .015f, MinGap = .002f, MaxGap = .015f, Side = .004f, KeepNearRim = .03f, BodyCell = .03f;
        static readonly Dictionary<Mesh, Mesh> cache = new Dictionary<Mesh, Mesh>();
        public static int LastDropped { get; private set; }

        static Vector3Int Key(Vector3 p, float cell) => new Vector3Int(Mathf.FloorToInt(p.x / cell), Mathf.FloorToInt(p.y / cell), Mathf.FloorToInt(p.z / cell));

        /// The garment's mesh without the inner wall away from its openings. `garment` is the top's renderer, `body` the hero's body
        /// renderer (to tell the inner wall, which faces the body, from the outer one).
        public static Mesh Prepare(SkinnedMeshRenderer garment, SkinnedMeshRenderer body, Mesh source)
        {
            if (!Enabled || !source || !garment || !body || !body.sharedMesh || !source.isReadable || !body.sharedMesh.isReadable) return source;
            if (cache.TryGetValue(source, out var done) && done) return done;
            var v = source.vertices; var nrm = source.normals;
            if (nrm == null || nrm.Length != v.Length) return source;

            // the body in the garment's mesh space (both meshes are in their rest pose)
            var toGarment = garment.transform.worldToLocalMatrix * body.transform.localToWorldMatrix;
            var bv = body.sharedMesh.vertices;
            var bodyGrid = new Dictionary<Vector3Int, List<Vector3>>();
            foreach (var b in bv) { var q = toGarment.MultiplyPoint3x4(b); var k = Key(q, BodyCell); if (!bodyGrid.TryGetValue(k, out var l)) bodyGrid[k] = l = new List<Vector3>(); l.Add(q); }
            Vector3? NearestBody(Vector3 p)
            {
                var k = Key(p, BodyCell); float best = float.MaxValue; Vector3 at = default; bool any = false;
                for (int r = 1; r <= 3 && !any; r++)
                    for (int x = -r; x <= r; x++) for (int y = -r; y <= r; y++) for (int z = -r; z <= r; z++)
                                if (bodyGrid.TryGetValue(k + new Vector3Int(x, y, z), out var l))
                                    foreach (var q in l) { float d = (q - p).sqrMagnitude; if (d < best) { best = d; at = q; any = true; } }
                return any ? at : (Vector3?)null;
            }

            // weld split vertices (seams, UV islands) so the two walls are found by position
            var weldOf = new int[v.Length]; var weldPos = new List<Vector3>(); var weldN = new List<Vector3>();
            var weldKey = new Dictionary<Vector3Int, int>();
            for (int i = 0; i < v.Length; i++)
            {
                var k = new Vector3Int(Mathf.RoundToInt(v[i].x / Weld), Mathf.RoundToInt(v[i].y / Weld), Mathf.RoundToInt(v[i].z / Weld));
                if (!weldKey.TryGetValue(k, out int w)) { w = weldPos.Count; weldKey[k] = w; weldPos.Add(v[i]); weldN.Add(Vector3.zero); }
                weldOf[i] = w; weldN[w] += nrm[i];
            }
            int n = weldPos.Count;
            for (int i = 0; i < n; i++) weldN[i] = weldN[i].normalized;
            var grid = new Dictionary<Vector3Int, List<int>>();
            for (int i = 0; i < n; i++) { var k = Key(weldPos[i], Cell); if (!grid.TryGetValue(k, out var l)) grid[k] = l = new List<int>(); l.Add(i); }

            // a point of the double wall has the other wall just behind it (along its own normal); the inner wall faces the body
            var paired = new bool[n]; var inner = new bool[n];
            for (int i = 0; i < n; i++)
            {
                var p = weldPos[i]; var nn = weldN[i]; var k = Key(p, Cell);
                for (int x = -1; x <= 1 && !paired[i]; x++) for (int y = -1; y <= 1 && !paired[i]; y++) for (int z = -1; z <= 1 && !paired[i]; z++)
                            if (grid.TryGetValue(k + new Vector3Int(x, y, z), out var l))
                                foreach (int j in l)
                                {
                                    var d = weldPos[j] - p; float along = Vector3.Dot(d, nn);
                                    if (along > -MinGap || along < -MaxGap) continue;
                                    if ((d - along * nn).magnitude < Side) { paired[i] = true; break; }
                                }
                if (!paired[i]) continue;
                var b = NearestBody(p);
                inner[i] = b.HasValue && Vector3.Dot(b.Value - p, nn) > 0;
            }
            // the rim: points of the garment that are not part of the double wall (the folded hems, cuffs, collar, placket edges)
            var rimGrid = new Dictionary<Vector3Int, List<Vector3>>();
            for (int i = 0; i < n; i++) if (!paired[i]) { var k = Key(weldPos[i], KeepNearRim); if (!rimGrid.TryGetValue(k, out var l)) rimGrid[k] = l = new List<Vector3>(); l.Add(weldPos[i]); }
            var hidden = new bool[n];
            for (int i = 0; i < n; i++)
            {
                if (!inner[i]) continue;
                var p = weldPos[i]; var k = Key(p, KeepNearRim); bool near = false;
                for (int x = -1; x <= 1 && !near; x++) for (int y = -1; y <= 1 && !near; y++) for (int z = -1; z <= 1 && !near; z++)
                            if (rimGrid.TryGetValue(k + new Vector3Int(x, y, z), out var l))
                                foreach (var q in l) if ((q - p).sqrMagnitude < KeepNearRim * KeepNearRim) { near = true; break; }
                hidden[i] = !near;
            }

            var mesh = UnityEngine.Object.Instantiate(source); mesh.name = source.name + " (shell)";
            int dropped = 0;
            for (int s = 0; s < source.subMeshCount; s++)
            {
                var tris = source.GetTriangles(s); var keep = new List<int>(tris.Length);
                for (int t = 0; t < tris.Length; t += 3)
                {
                    if (hidden[weldOf[tris[t]]] && hidden[weldOf[tris[t + 1]]] && hidden[weldOf[tris[t + 2]]]) { dropped++; continue; }
                    keep.Add(tris[t]); keep.Add(tris[t + 1]); keep.Add(tris[t + 2]);
                }
                mesh.SetTriangles(keep, s, false);
            }
            mesh.bounds = source.bounds;
            LastDropped = dropped;
            if (dropped == 0) { if (Application.isPlaying) UnityEngine.Object.Destroy(mesh); else UnityEngine.Object.DestroyImmediate(mesh); cache[source] = source; return source; }
            Debug.Log($"[HeroGarmentShell] {source.name}: {dropped} inner-wall triangles of {source.triangles.Length / 3} dropped ({n} welded points, {Count(paired)} in the double wall, {Count(inner)} inner)");
            cache[source] = mesh;
            return mesh;
        }

        static int Count(bool[] a) { int c = 0; foreach (var b in a) if (b) c++; return c; }
    }
}
