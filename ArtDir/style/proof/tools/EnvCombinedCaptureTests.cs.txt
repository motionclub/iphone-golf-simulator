        [UnityTest, Timeout(1500000)] public IEnumerator B_Golf()
        {
            Directory.CreateDirectory(Out); UnityEngine.Random.InitState(141026);
            RoundEvents.Enabled = true;
            yield return SceneManager.LoadSceneAsync("Golf", LoadSceneMode.Single); yield return null;
            var cam = Camera.main;
            var game = Object.FindFirstObjectByType<GolfGame>(); game.InstantReplays = false;
            game.SetRoundEvents(true, true, true);
            game.ChooseHoles(0); game.Play(); yield return null;
            var hud = Object.FindFirstObjectByType<GolfArcade.UI.Hud>().GetComponent<Canvas>();
            var shots = new (int hole, Weather w, bool views)[] {
                (7, Weather.Clear, true), (12, Weather.Clear, true), (14, Weather.Clear, true), (20, Weather.Clear, true),
                (17, Weather.Clear, true), (16, Weather.Clear, true), (22, Weather.Clear, true),
                (7, Weather.Gusty, false), (7, Weather.Rain, false), (7, Weather.Storm, false), (7, Weather.Fog, false),
                (20, Weather.Rain, false), (20, Weather.Storm, false), (20, Weather.Fog, false) };
            if(System.Environment.GetEnvironmentVariable("ENV_ALL")=="1")
                shots=GolfArcade.Course.Course.All().SelectMany(c=>c.Holes).Select(h=>(h.Number,Weather.Clear,true))
                    .Concat(new[]{7,17,20,22}.SelectMany(h=>new[]{Weather.Gusty,Weather.Rain,Weather.Storm,Weather.Fog}.Select(w=>(h,w,false)))).ToArray();
            foreach (var (number, w, views) in shots)
            {
                GolfGame.ForcedWeather = w;
                game.JumpToHole(number);
                yield return Wait(() => { if(game.Current != GolfGame.State.Aim) game.SkipPresentation(); return game.Current == GolfGame.State.Aim; }, 60);
                yield return new WaitForSecondsRealtime(1.5f);
                Assert.AreEqual(number,game.CurrentHole.Number,"Requested hole must be active");
                Assert.AreEqual(w,game.CurrentWeather,"Requested weather must be active");
                cam.aspect = 16f / 9f;
                string n = $"golf_{number:00}_{w.ToString().ToLower()}";
                ProofCapture($"{Out}/{n}_1_tee_gameplay.png", 1920, 1080);
                hud.enabled = false;
                var hole = game.CurrentHole;
                var tee = HoleView.ToWorld(hole.Tee); var pin = HoleView.ToWorld(hole.Pin);
                var along = pin - tee; along.y = 0; float len = along.magnitude; along /= len;
                var right = Vector3.Cross(Vector3.up, along);
                var list = views ? new (string, Vector3, Vector3)[] {
                    ("2_approach", tee + along * len * .6f + Vector3.up * 3f, pin + Vector3.up),
                    ("3_green", pin - along * 30 + right * 12 + Vector3.up * 6, pin + Vector3.up * 1.5f),
                    ("4_overview", tee - along * 50 + Vector3.up * (len * .35f + 30), (tee + pin) / 2) }
                  : new (string, Vector3, Vector3)[] { ("2_approach", tee + along * len * .6f + Vector3.up * 3f, pin + Vector3.up) };
                foreach (var (vn, at, look) in list)
                {
                    game.enabled = false; cam.transform.position = at; cam.transform.LookAt(look);
                    yield return null;
                    ProofCapture($"{Out}/{n}_{vn}.png", 1920, 1080);
                    game.enabled = true;
                }
                hud.enabled = true;
            }
            GolfGame.ForcedWeather = null;
        }

        static void ProofCapture(string path,int width,int height)
        {
            GameCapture.Save(path,width,height);
            RecordEffectiveState(path);
            if(System.Environment.GetEnvironmentVariable("ENV_TONE_PAIR")!="1" ||
                !(path.EndsWith("tennis_resort_2_close.png") || path.EndsWith("golf_07_clear_1_tee_gameplay.png")))return;
            var tones=new System.Collections.Generic.List<(Tonemapping component,TonemappingMode mode)>();
            foreach(var volume in Object.FindObjectsByType<Volume>(FindObjectsSortMode.None))
                if(volume.profile.TryGet(out Tonemapping tone))tones.Add((tone,tone.mode.value));
            try {
                foreach(var mode in new[]{TonemappingMode.Neutral,TonemappingMode.ACES}) {
                    foreach(var t in tones)t.component.mode.Override(mode);
                    GameCapture.Save(Path.Combine(Out,"tone-pair",Path.GetFileNameWithoutExtension(path)+"_"+mode+".png"),width,height);
                }
            } finally { foreach(var t in tones)t.component.mode.Override(t.mode); }
        }

        [System.Serializable] class EffectiveState {
            public string shot,tier,device,mapper,courtMaterials;public Color probeUp,probeSide;public int stage;public float sun,shadow,fogNear,fogFar,heroFill,heroRim,heroAmbientMin,heroAmbientMax,wet,storm;
            public Color sky,equator,ground,fog;public Vector4 layers;public bool depthOfField,bloom;
            public Vector3 cameraPosition;public bool hasGroundSample;public float groundBelowCamera,cameraGroundClearance;
        }
        static void RecordEffectiveState(string path) {
            var sun=RenderSettings.sun;
            var state=new EffectiveState {shot=Path.GetFileName(path),tier=TennisQuality.Current.ToString(),device=SystemInfo.deviceModel,
                stage=GolfArcade.EnvironmentArt.EnvironmentLook.Stage,sun=sun?sun.intensity:0,shadow=sun?sun.shadowStrength:0,
                fogNear=RenderSettings.fogStartDistance,fogFar=RenderSettings.fogEndDistance,sky=RenderSettings.ambientSkyColor,
                equator=RenderSettings.ambientEquatorColor,ground=RenderSettings.ambientGroundColor,fog=RenderSettings.fogColor,
                heroFill=Shader.GetGlobalFloat("_HeroPresentationFill"),heroRim=Shader.GetGlobalFloat("_HeroAnalyticRimStrength"),
                heroAmbientMin=Shader.GetGlobalFloat("_HeroAmbientMin"),heroAmbientMax=Shader.GetGlobalFloat("_HeroAmbientMax"),
                wet=Shader.GetGlobalFloat("_EnvironmentWetness"),storm=Shader.GetGlobalFloat("_EnvironmentStorm"),layers=Shader.GetGlobalVector("_EnvironmentDepthLayers")};
            foreach(var volume in Object.FindObjectsByType<Volume>(FindObjectsSortMode.None)) {
                if(volume.profile.TryGet(out Tonemapping tone))state.mapper=tone.mode.value.ToString();
                if(volume.profile.TryGet(out DepthOfField dof))state.depthOfField=dof.active;
                if(volume.profile.TryGet(out Bloom bloom))state.bloom=bloom.active;
            }
            var sh=new Color[2];RenderSettings.ambientProbe.Evaluate(new[]{Vector3.up,Vector3.forward},sh);state.probeUp=sh[0];state.probeSide=sh[1];
            var game=Object.FindFirstObjectByType<GolfGame>();var camera=Camera.main;
            if(camera)state.cameraPosition=camera.transform.position;
            if(game && game.CurrentHole!=null && camera) {
                var at=new CoursePoint(camera.transform.position.x,camera.transform.position.z);
                state.groundBelowCamera=(float)(game.CurrentHole.Ground!=null?game.CurrentHole.Ground(at):HoleView.GroundHeight(at));
                state.cameraGroundClearance=camera.transform.position.y-state.groundBelowCamera;state.hasGroundSample=true;
            }
            state.courtMaterials=string.Join(";",Object.FindObjectsByType<Renderer>(FindObjectsSortMode.None).SelectMany(r=>r.sharedMaterials)
                .Where(m=>m && m.name.StartsWith("TropicalV3_002")).Distinct().Select(m=>m.name+":"+ColorUtility.ToHtmlStringRGB(m.GetColor("_BaseColor"))));
            File.AppendAllText(Path.Combine(Out,"effective-state.jsonl"),JsonUtility.ToJson(state)+"\n");
        }
        static void Still(Camera cam, string path, int w = 1920, int h = 1080)
        {
            var rt = RenderTexture.GetTemporary(w, h, 24, RenderTextureFormat.ARGB32); rt.antiAliasing = 4;
            var prevT = cam.targetTexture; var prevA = cam.aspect;
            cam.targetTexture = rt; cam.aspect = w / (float)h; cam.Render(); cam.targetTexture = prevT; cam.aspect = prevA;
            var prev = RenderTexture.active; RenderTexture.active = rt;
            var tex = new Texture2D(w, h, TextureFormat.RGB24, false); tex.ReadPixels(new Rect(0, 0, w, h), 0, 0); tex.Apply();
            RenderTexture.active = prev; RenderTexture.ReleaseTemporary(rt);
            File.WriteAllBytes(path, tex.EncodeToPNG()); Object.Destroy(tex);RecordEffectiveState(path);
        }
