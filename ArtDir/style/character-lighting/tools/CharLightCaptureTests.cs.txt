    /// Scratch-only capture: character lighting audit. Writes PNGs + boxes.csv (hero screen box per shot).
    public class CharLightCaptureTests
    {
        static string Out => System.Environment.GetEnvironmentVariable("AUDIT_OUT") ?? "Library/Captures/charlight";
        static IEnumerator Wait(System.Func<bool> ok, float s) { float t = 0; while (!ok() && t < s) { t += Time.unscaledDeltaTime; yield return null; } }
        static void Still(Camera cam, string name, Transform hero)
        {
            int w = 1600, h = 900;
            var rt = RenderTexture.GetTemporary(w, h, 24, RenderTextureFormat.ARGB32, RenderTextureReadWrite.Default, 4);
            var prevT = cam.targetTexture; var prevA = cam.aspect;
            cam.targetTexture = rt; cam.aspect = w / (float)h; cam.Render();
            // hero screen box from renderers
            var b = new Bounds(hero.position + Vector3.up, Vector3.zero); bool any = false;
            foreach (var r in hero.GetComponentsInChildren<Renderer>()) { if (!r.enabled || !r.gameObject.activeInHierarchy || r is ParticleSystemRenderer) continue; if (!any) { b = r.bounds; any = true; } else b.Encapsulate(r.bounds); }
            float x0 = 1, y0 = 1, x1 = 0, y1 = 0;
            for (int i = 0; i < 8; i++) { var c = b.center + Vector3.Scale(b.extents, new Vector3((i & 1) * 2 - 1, (i & 2) - 1, ((i & 4) >> 1) - 1)); var v = cam.WorldToViewportPoint(c); if (v.z <= 0) continue; x0 = Mathf.Min(x0, v.x); x1 = Mathf.Max(x1, v.x); y0 = Mathf.Min(y0, v.y); y1 = Mathf.Max(y1, v.y); }
            var head = cam.WorldToViewportPoint(b.center + Vector3.up * b.extents.y * .78f);
            cam.targetTexture = prevT; cam.aspect = prevA;
            var prev = RenderTexture.active; RenderTexture.active = rt;
            var tex = new Texture2D(w, h, TextureFormat.RGB24, false); tex.ReadPixels(new Rect(0, 0, w, h), 0, 0); tex.Apply();
            RenderTexture.active = prev; RenderTexture.ReleaseTemporary(rt);
            File.WriteAllBytes($"{Out}/{name}.png", tex.EncodeToPNG()); Object.Destroy(tex);
            // hero-only mask pass (layer 31, black background, unlit-independent: any non-black pixel = hero)
            var rs = hero.GetComponentsInChildren<Renderer>(); var layers = new int[rs.Length];
            for (int i = 0; i < rs.Length; i++) { layers[i] = rs[i].gameObject.layer; rs[i].gameObject.layer = 31; }
            var mask = cam.cullingMask; var cf = cam.clearFlags; var bg = cam.backgroundColor;
            var rt2 = RenderTexture.GetTemporary(w, h, 24, RenderTextureFormat.ARGB32);
            cam.cullingMask = 1 << 31; cam.clearFlags = CameraClearFlags.SolidColor; cam.backgroundColor = Color.black;
            cam.targetTexture = rt2; cam.aspect = w / (float)h; cam.Render(); cam.targetTexture = prevT; cam.aspect = prevA;
            cam.cullingMask = mask; cam.clearFlags = cf; cam.backgroundColor = bg;
            for (int i = 0; i < rs.Length; i++) rs[i].gameObject.layer = layers[i];
            RenderTexture.active = rt2; var t2 = new Texture2D(w, h, TextureFormat.RGB24, false); t2.ReadPixels(new Rect(0, 0, w, h), 0, 0); t2.Apply();
            RenderTexture.active = prev; RenderTexture.ReleaseTemporary(rt2);
            File.WriteAllBytes($"{Out}/{name}.mask.png", t2.EncodeToPNG()); Object.Destroy(t2);
            File.AppendAllText($"{Out}/boxes.csv", $"{name},{x0:0.0000},{1 - y1:0.0000},{x1:0.0000},{1 - y0:0.0000},{head.x:0.0000},{1 - head.y:0.0000}\n");
        }
        static void Close(Camera cam, Transform hero, string name, bool sideOn = false)
        {
            var f = hero.forward; f.y = 0; f.Normalize(); var r = Vector3.Cross(Vector3.up, f);
            var p = hero.position; var keep = (cam.transform.position, cam.transform.rotation, cam.fieldOfView);
            cam.transform.position = p + f * 2.4f + r * 1.3f + Vector3.up * 1.55f; cam.transform.LookAt(p + Vector3.up * 1.15f); cam.fieldOfView = 34;
            Still(cam, name, hero);
            cam.transform.SetPositionAndRotation(keep.position, keep.rotation); cam.fieldOfView = keep.fieldOfView;
        }

        [UnityTest, Timeout(1500000)] public IEnumerator A_Tennis()
        {
            Directory.CreateDirectory(Out);
            foreach (var v in new[] { TennisVenueKind.Resort, TennisVenueKind.Skyscraper, TennisVenueKind.Volcano })
            foreach (var female in new[] { false, true })
            {
                TennisVenue.Selected = v;
                yield return SceneManager.LoadSceneAsync("Tennis", LoadSceneMode.Single); yield return null;
                var game = Object.FindFirstObjectByType<TennisGame>(); game.ManualSimulation = true; game.NativeControlled = true;
                game.ConfigureMatch(TennisGame.Mode.Training, null, null, null);
                game.SelectCharacter(female);
                for (int i = 0; i < 360; i++) game.Step(1f / 120);
                for (int i = 0; i < 40; i++) yield return null;
                game.Player.transform.position = new Vector3(0, .035f, TennisRules.BaselineZ);
                yield return null; yield return null;
                var cam = game.GameplayCamera ? game.GameplayCamera : Camera.main;
                var p = game.Player.transform.position;
                string n = $"MatchHero_{(female ? "Female" : "Male")}__tennis_{v.ToString().ToLower()}";
                cam.transform.position = new Vector3(p.x, 2.5f, p.z - 3.5f); cam.transform.LookAt(new Vector3(0, 1f, 3f)); cam.fieldOfView = 46;
                Still(cam, n + "__a_gameplay", game.Player.transform);
                var hero = game.Player.transform; var f = Vector3.back; // face camera side: use world -z (toward player's back is +z net); close-up from the net side
                var keep = (cam.transform.position, cam.transform.rotation);
                cam.transform.position = p + new Vector3(1.3f, 1.55f, 2.4f); cam.transform.LookAt(p + Vector3.up * 1.15f); cam.fieldOfView = 34;
                Still(cam, n + "__b_close", hero);
                cam.transform.SetPositionAndRotation(keep.position, keep.rotation);
            }
        }

        [UnityTest, Timeout(2400000)] public IEnumerator B_Golf()
        {
            Directory.CreateDirectory(Out);
            yield return SceneManager.LoadSceneAsync("Golf", LoadSceneMode.Single); yield return null;
            var cam = Camera.main;
            RoundEvents.Enabled = true;
            var game = Object.FindFirstObjectByType<GolfGame>(); game.InstantReplays = false;
            game.SetRoundEvents(true, true, true);
            game.ChooseHoles(0); game.Play(); yield return null;
            var shots = new (int hole, Weather w)[] { (7, Weather.Clear), (16, Weather.Clear), (17, Weather.Clear), (20, Weather.Clear), (7, Weather.Rain), (7, Weather.Storm), (7, Weather.Fog) };
            foreach (var female in new[] { false, true })
            foreach (var (number, w) in shots)
            {
                GolferStyle.Body = female ? GolferStyle.BodyKind.Female : GolferStyle.BodyKind.Male; game.RestyleGolfer();
                GolfGame.ForcedWeather = w;
                game.JumpToHole(number);
                yield return Wait(() => game.Current == GolfGame.State.Aim, 60);
                yield return new WaitForSecondsRealtime(1.5f);
                var golfer = Object.FindFirstObjectByType<GolferView>().transform;
                string n = $"GolfKitHero_{(female ? "Female" : "Male")}__golf_{number:00}_{w.ToString().ToLower()}";
                Still(cam, n + "__a_gameplay", golfer);
                game.enabled = false; yield return null;
                var to = cam.transform.position - golfer.position; to.y = 0; to.Normalize(); var r = Vector3.Cross(Vector3.up, to);
                var keep = (cam.transform.position, cam.transform.rotation, cam.fieldOfView);
                // 3/4 front: rotate 90 deg from the tee cam (which is behind) toward the golfer's face side
                var face = golfer.forward; face.y = 0; face.Normalize();
                cam.transform.position = golfer.position + face * 2.4f + to * 1.2f + Vector3.up * 1.55f; cam.transform.LookAt(golfer.position + Vector3.up * 1.1f); cam.fieldOfView = 34;
                Still(cam, n + "__b_close", golfer);
                cam.transform.SetPositionAndRotation(keep.position, keep.rotation); cam.fieldOfView = keep.fieldOfView;
                game.enabled = true;
            }
            GolfGame.ForcedWeather = null;
        }

        [UnityTest, Timeout(900000)] public IEnumerator C_Hub()
        {
            Directory.CreateDirectory(Out);
            foreach (var female in new[] { false, true })
            {
                HubWorld.EditorDriven = true;
                var host = new GameObject("NativeSportsSession"); Object.DontDestroyOnLoad(host);
                var bridge = host.AddComponent<NativeSportsSession>();
                bridge.Receive(JsonUtility.ToJson(new NativeSportsSession.Message { version = 1, session = "hub-test" + female, action = "start", sport = "hub", female = female, touch = true, shirt = "E8505B", shorts = "24345E", accent = "FFFFFF" }));
                for (int i = 0; i < 900 && !bridge.Ready; i++) yield return null;
                yield return new WaitForSeconds(4f);
                var hub = HubWorld.Find(); var p = hub.Player; var cam = hub.Camera;
                p.Teleport(HubLayout.Spawn, 0); yield return new WaitForSeconds(1.5f);
                string n = $"MatchHero_{(female ? "Female" : "Male")}__hub";
                Still(cam, n + "__a_gameplay", p.transform);
                var face = p.transform.forward; face.y = 0; face.Normalize(); var r = Vector3.Cross(Vector3.up, face);
                cam.transform.position = p.transform.position + face * 2.4f + r * 1.3f + Vector3.up * 1.55f; cam.transform.LookAt(p.transform.position + Vector3.up * 1.15f); cam.fieldOfView = 34;
                Still(cam, n + "__b_close", p.transform);
                HubWorld.EditorDriven = false; Object.DestroyImmediate(host);
                yield return SceneManager.LoadSceneAsync("Tennis", LoadSceneMode.Single);
                yield return new WaitForSeconds(1f);
            }
        }
    }
}
