        static string Out => System.Environment.GetEnvironmentVariable("AUDIT_OUT") ?? "Library/Captures/charlight";
        static IEnumerator Wait(System.Func<bool> ok, float s) { float t = 0; while (!ok() && t < s) { t += Time.unscaledDeltaTime; yield return null; } }
        static void Still(Camera cam, string name, Transform hero)
        {
            int w = 1600, h = 900;
            var rt = RenderTexture.GetTemporary(w, h, 24, RenderTextureFormat.ARGB32, RenderTextureReadWrite.Default, 4);
            var prevT = cam.targetTexture; var prevA = cam.aspect;
            cam.targetTexture = rt; cam.aspect = w / (float)h; cam.Render();
            float lighting=Shader.GetGlobalFloat("_HeroLightingEnabled"),fill=Shader.GetGlobalFloat("_HeroPresentationFill");
            // Both images use the same geometry, pose, camera, post, weather and light state.
            if(lighting>.5f) {
                Shader.SetGlobalFloat("_HeroLightingEnabled",0);Shader.SetGlobalFloat("_HeroPresentationFill",.12f);
                cam.Render();var old=RenderTexture.active;RenderTexture.active=rt;
                var baseline=new Texture2D(w,h,TextureFormat.RGB24,false);baseline.ReadPixels(new Rect(0,0,w,h),0,0);baseline.Apply();
                Directory.CreateDirectory(Out+"/paired-before");File.WriteAllBytes($"{Out}/paired-before/{name}.png",baseline.EncodeToPNG());Object.Destroy(baseline);RenderTexture.active=old;
                Shader.SetGlobalFloat("_HeroLightingEnabled",lighting);Shader.SetGlobalFloat("_HeroPresentationFill",fill);cam.Render();
            }
            // hero screen box from renderers
            var b = new Bounds(hero.position + Vector3.up, Vector3.zero); bool any = false;
            foreach (var r in hero.GetComponentsInChildren<Renderer>()) { if (!r.enabled || !r.gameObject.activeInHierarchy || r is ParticleSystemRenderer) continue; if (!any) { b = r.bounds; any = true; } else b.Encapsulate(r.bounds); }
            float x0 = 1, y0 = 1, x1 = 0, y1 = 0;
            for (int i = 0; i < 8; i++) { var c = b.center + Vector3.Scale(b.extents, new Vector3((i & 1) * 2 - 1, (i & 2) - 1, ((i & 4) >> 1) - 1)); var v = cam.WorldToViewportPoint(c); if (v.z <= 0) continue; x0 = Mathf.Min(x0, v.x); x1 = Mathf.Max(x1, v.x); y0 = Mathf.Min(y0, v.y); y1 = Mathf.Max(y1, v.y); }
            var head = cam.WorldToViewportPoint(b.center + Vector3.up * b.extents.y * .78f);
            cam.targetTexture = prevT; cam.aspect = prevA;
            var prev = RenderTexture.active; RenderTexture.active = rt;
            var tex = new Texture2D(w, h, TextureFormat.RGB24, false); tex.ReadPixels(new Rect(0, 0, w, h), 0, 0); tex.Apply();
            RenderTexture.active = prev; RenderTexture.ReleaseTemporary(rt);
            File.WriteAllBytes($"{Out}/{name}.png", tex.EncodeToPNG()); Object.Destroy(tex);
            FreezeForLookDev(name,hero);
            CharLightDiagnostics.Capture(cam, $"{Out}/{name}.mask.png", hero);
            File.AppendAllText($"{Out}/boxes.csv", $"{name},{x0:0.0000},{1 - y1:0.0000},{x1:0.0000},{1 - y0:0.0000},{head.x:0.0000},{1 - head.y:0.0000}\n");
        }
        static void Close(Camera cam, Transform hero, string name, bool sideOn = false)
        {
            var f = hero.forward; f.y = 0; f.Normalize(); var r = Vector3.Cross(Vector3.up, f);
            var p = hero.position; var keep = (cam.transform.position, cam.transform.rotation, cam.fieldOfView);
            cam.transform.position = p + f * 2.4f + r * 1.3f + Vector3.up * 1.55f; cam.transform.LookAt(p + Vector3.up * 1.15f); cam.fieldOfView = 34;
            Still(cam, name, hero);
            cam.transform.SetPositionAndRotation(keep.position, keep.rotation); cam.fieldOfView = keep.fieldOfView;
        }

        [UnityTest, Timeout(2400000)] public IEnumerator B_Golf()
        {
            Directory.CreateDirectory(Out);
            yield return SceneManager.LoadSceneAsync("Golf", LoadSceneMode.Single); yield return null;
            var cam = Camera.main;
            RoundEvents.Enabled = true;
            var game = Object.FindFirstObjectByType<GolfGame>(); game.InstantReplays = false;
            game.SetRoundEvents(true, true, true);
            game.ChooseHoles(0); game.Play(); yield return null;
            var shots = new (int hole, Weather w)[] { (7, Weather.Clear), (16, Weather.Clear), (17, Weather.Clear), (20, Weather.Clear), (7, Weather.Rain), (7, Weather.Storm), (7, Weather.Fog) };
            if(System.Environment.GetEnvironmentVariable("CHARLIGHT_QUICK")=="1")shots=new[] {(7,Weather.Clear)};
            else if(System.Environment.GetEnvironmentVariable("CHARLIGHT_ALL")=="1")shots=GolfArcade.Course.Course.All().SelectMany(c=>c.Holes).SelectMany(h=>System.Enum.GetValues(typeof(Weather)).Cast<Weather>().Select(w=>(h.Number,w))).ToArray();
            foreach (var female in new[] { false, true })
            foreach (var (number, w) in shots)
            {
                GolferStyle.Body = female ? GolferStyle.BodyKind.Female : GolferStyle.BodyKind.Male; game.RestyleGolfer();
                GolfGame.ForcedWeather = w;
                game.JumpToHole(number);
                yield return Wait(() => { if(System.Environment.GetEnvironmentVariable("CHARLIGHT_ALL")=="1" && game.Current!=GolfGame.State.Aim)game.SkipPresentation(); return game.Current==GolfGame.State.Aim; }, 60);
                yield return new WaitForSecondsRealtime(1.5f);
                Assert.AreEqual(number,game.CurrentHole.Number,"Capture must use the requested hole");
                Assert.AreEqual(w,Object.FindFirstObjectByType<WeatherView>().Weather,"Capture must use requested weather");
                var golfer = Object.FindFirstObjectByType<GolferView>().transform;
                string n = $"GolfKitHero_{(female ? "Female" : "Male")}__golf_{number:00}_{w.ToString().ToLower()}";
                Still(cam, n + "__a_gameplay", golfer);
                game.enabled = false; yield return null;
                var to = cam.transform.position - golfer.position; to.y = 0; to.Normalize(); var r = Vector3.Cross(Vector3.up, to);
                var keep = (cam.transform.position, cam.transform.rotation, cam.fieldOfView);
                // 3/4 front: rotate 90 deg from the tee cam (which is behind) toward the golfer's face side
                var face = golfer.forward; face.y = 0; face.Normalize();
                cam.transform.position = golfer.position + face * 2.4f + to * 1.2f + Vector3.up * 1.55f; cam.transform.LookAt(golfer.position + Vector3.up * 1.1f); cam.fieldOfView = 34;
                Still(cam, n + "__b_close", golfer);
                cam.transform.SetPositionAndRotation(keep.position, keep.rotation); cam.fieldOfView = keep.fieldOfView;
                game.enabled = true;
            }
            GolfGame.ForcedWeather = null;
        }

        static readonly System.Collections.Generic.Dictionary<string,GameObject> frozen=new();
        static void FreezeForLookDev(string name,Transform source) {
            if(!name.EndsWith("__a_gameplay") || !(name.Contains("tennis_resort") || name.Contains("golf_07_clear")))return;
            string key=name.Split(new[]{"__"},System.StringSplitOptions.None)[0];if(frozen.ContainsKey(key))return;
            var root=new GameObject("Frozen lookdev "+key);root.transform.SetPositionAndRotation(source.position,source.rotation);Object.DontDestroyOnLoad(root);
            foreach(var r in source.GetComponentsInChildren<Renderer>()) {
                if(!r.enabled || !r.gameObject.activeInHierarchy || !(r is SkinnedMeshRenderer || r is MeshRenderer))continue;
                Mesh mesh=null;
                if(r is SkinnedMeshRenderer skin) { mesh=new Mesh();skin.BakeMesh(mesh); }
                else if(r.GetComponent<MeshFilter>())mesh=Object.Instantiate(r.GetComponent<MeshFilter>().sharedMesh);
                if(!mesh)continue;
                var go=new GameObject(r.name);go.transform.SetParent(root.transform,false);go.transform.SetPositionAndRotation(r.transform.position,r.transform.rotation);go.transform.localScale=r.transform.lossyScale;
                go.AddComponent<MeshFilter>().sharedMesh=mesh;var mr=go.AddComponent<MeshRenderer>();mr.sharedMaterials=r.sharedMaterials.Select(m=>m?new Material(m):null).ToArray();mr.shadowCastingMode=r.shadowCastingMode;mr.receiveShadows=r.receiveShadows;mr.renderingLayerMask=r.renderingLayerMask;
            }
            root.SetActive(false);frozen.Add(key,root);
        }
        [UnityTest,Timeout(900000)] public IEnumerator D_FourHeroesBothEnvironments() {
            Assert.AreEqual(4,frozen.Count,"Four hero snapshots required from A and B");
            foreach(bool golfScene in new[]{false,true}) {
                Camera cam;Transform player;
                if(!golfScene) {
                    TennisVenue.Selected=TennisVenueKind.Resort;yield return SceneManager.LoadSceneAsync("Tennis");yield return null;
                    var g=Object.FindFirstObjectByType<TennisGame>();g.ManualSimulation=true;g.NativeControlled=true;g.ConfigureMatch(TennisGame.Mode.Training,null,null,null);
                    for(int i=0;i<40;i++)yield return null;
                    player=g.Player.transform;player.position=new Vector3(0,.035f,TennisRules.BaselineZ);cam=g.GameplayCamera?g.GameplayCamera:Camera.main;
                } else {
                    yield return SceneManager.LoadSceneAsync("Golf");yield return null;
                    var g=Object.FindFirstObjectByType<GolfGame>();g.InstantReplays=false;g.SetRoundEvents(true,true,true);g.ChooseHoles(0);g.Play();GolfGame.ForcedWeather=Weather.Clear;g.JumpToHole(7);
                    yield return Wait(()=>g.Current==GolfGame.State.Aim,60);yield return new WaitForSecondsRealtime(1.5f);player=Object.FindFirstObjectByType<GolferView>().transform;cam=Camera.main;g.enabled=false;
                }
                foreach(var r in player.GetComponentsInChildren<Renderer>())r.enabled=false;
                foreach(var entry in frozen) {
                    var hero=entry.Value.transform;hero.position=player.position;hero.rotation=Quaternion.identity;entry.Value.SetActive(true);
                    // Frozen production poses let both environments light all four kits without changing any rig/clip asset.
                    cam.transform.position=hero.position+new Vector3(1.3f,1.55f,2.4f);cam.transform.LookAt(hero.position+Vector3.up*1.1f);cam.fieldOfView=34;
                    Still(cam,"lookdev_"+(golfScene?"golf07":"resort")+"__"+entry.Key+"__b_close",hero);
                    entry.Value.SetActive(false);
                }
            }
            foreach(var go in frozen.Values){foreach(var r in go.GetComponentsInChildren<MeshRenderer>(true)){foreach(var m in r.sharedMaterials)Object.Destroy(m);Object.Destroy(r.GetComponent<MeshFilter>().sharedMesh);}Object.Destroy(go);}frozen.Clear();GolfGame.ForcedWeather=null;
        }
